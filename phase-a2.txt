import crypto from "node:crypto";
import { cookies } from "next/headers";

import { prisma } from "@/lib/prisma";

const COOKIE_NAME =
  process.env.SESSION_COOKIE_NAME || "attendance_session";

const SESSION_DAYS = 7;

function hashToken(token: string): string {
  return crypto
    .createHash("sha256")
    .update(token)
    .digest("hex");
}

function generateToken(): string {
  return crypto.randomBytes(32).toString("hex");
}

export async function createSession(userId: string) {
  const token = generateToken();
  const tokenHash = hashToken(token);

  const expiresAt = new Date(
    Date.now() + SESSION_DAYS * 24 * 60 * 60 * 1000,
  );

  await prisma.session.create({
    data: {
      userId,
      tokenHash,
      expiresAt,
    },
  });

  const cookieStore = await cookies();

  cookieStore.set({
    name: COOKIE_NAME,
    value: token,
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    expires: expiresAt,
    path: "/",
  });

  return {
    expiresAt,
  };
}

export async function getCurrentSession() {
  const cookieStore = await cookies();

  const token = cookieStore.get(COOKIE_NAME)?.value;

  if (!token) {
    return null;
  }

  const tokenHash = hashToken(token);

  const session = await prisma.session.findUnique({
    where: {
      tokenHash,
    },
    include: {
      user: true,
    },
  });

  if (!session) {
    return null;
  }

  if (session.expiresAt <= new Date()) {
    await prisma.session.delete({
      where: {
        id: session.id,
      },
    });

    cookieStore.delete(COOKIE_NAME);

    return null;
  }

  if (!session.user.isActive) {
    await prisma.session.delete({
      where: {
        id: session.id,
      },
    });

    cookieStore.delete(COOKIE_NAME);

    return null;
  }

  return session;
}

export async function deleteCurrentSession() {
  const cookieStore = await cookies();

  const token = cookieStore.get(COOKIE_NAME)?.value;

  if (token) {
    const tokenHash = hashToken(token);

    await prisma.session.deleteMany({
      where: {
        tokenHash,
      },
    });
  }

  cookieStore.delete(COOKIE_NAME);
}

const TIME_ZONE =
  process.env.APP_TIMEZONE || "Asia/Tehran";

type DateParts = {
  year: number;
  month: number;
  day: number;
};

function getDateParts(date: Date): DateParts {
  const formatter =
    new Intl.DateTimeFormat("en-US", {
      timeZone: TIME_ZONE,
      year: "numeric",
      month: "2-digit",
      day: "2-digit",
    });

  const parts =
    formatter.formatToParts(date);

  return {
    year: Number(
      parts.find(
        (part) => part.type === "year",
      )?.value,
    ),

    month: Number(
      parts.find(
        (part) => part.type === "month",
      )?.value,
    ),

    day: Number(
      parts.find(
        (part) => part.type === "day",
      )?.value,
    ),
  };
}

function isValidDateParts(
  parts: DateParts,
): boolean {
  if (
    !Number.isInteger(parts.year) ||
    !Number.isInteger(parts.month) ||
    !Number.isInteger(parts.day)
  ) {
    return false;
  }

  if (
    parts.month < 1 ||
    parts.month > 12 ||
    parts.day < 1 ||
    parts.day > 31
  ) {
    return false;
  }

  /*
   * Date.UTC به‌صورت خودکار تاریخ‌های
   * نامعتبر مثل 2026-02-31 را normalize می‌کند.
   *
   * با مقایسه مجدد اجزای تاریخ، مطمئن
   * می‌شویم تاریخ واقعاً در تقویم میلادی
   * معتبر است.
   */
  const probe = new Date(
    Date.UTC(
      parts.year,
      parts.month - 1,
      parts.day,
      12,
      0,
      0,
      0,
    ),
  );

  return (
    probe.getUTCFullYear() ===
      parts.year &&
    probe.getUTCMonth() ===
      parts.month - 1 &&
    probe.getUTCDate() ===
      parts.day
  );
}

function addCalendarDays(
  parts: DateParts,
  amount: number,
): DateParts {
  const probe = new Date(
    Date.UTC(
      parts.year,
      parts.month - 1,
      parts.day + amount,
      12,
      0,
      0,
      0,
    ),
  );

  return {
    year:
      probe.getUTCFullYear(),

    month:
      probe.getUTCMonth() + 1,

    day:
      probe.getUTCDate(),
  };
}

function getTimeZoneOffsetMs(
  date: Date,
): number {
  const formatter =
    new Intl.DateTimeFormat("en-US", {
      timeZone: TIME_ZONE,
      timeZoneName: "shortOffset",
    });

  const parts =
    formatter.formatToParts(date);

  const value =
    parts.find(
      (part) =>
        part.type === "timeZoneName",
    )?.value || "GMT";

  if (
    value === "GMT" ||
    value === "UTC"
  ) {
    return 0;
  }

  const match =
    value.match(
      /^GMT([+-])(\d{1,2})(?::(\d{2}))?$/,
    );

  if (!match) {
    return 0;
  }

  const sign =
    match[1] === "+" ? 1 : -1;

  const hours =
    Number(match[2]);

  const minutes =
    Number(match[3] || 0);

  return (
    sign *
    (hours * 60 + minutes) *
    60 *
    1000
  );
}

function getLocalMidnightUtc(
  parts: DateParts,
): Date {
  /*
   * ابتدا یک حدس در ظهر UTC می‌سازیم.
   * استفاده از ظهر به‌جای نیمه‌شب احتمال
   * برخورد با مرزهای تغییر offset را کمتر می‌کند.
   */
  const utcGuess =
    new Date(
      Date.UTC(
        parts.year,
        parts.month - 1,
        parts.day,
        12,
        0,
        0,
        0,
      ),
    );

  const firstOffset =
    getTimeZoneOffsetMs(
      utcGuess,
    );

  let result =
    new Date(
      Date.UTC(
        parts.year,
        parts.month - 1,
        parts.day,
        0,
        0,
        0,
        0,
      ) - firstOffset,
    );

  /*
   * یک بار دیگر offset را از خود timestamp
   * نهایی می‌گیریم تا اگر timezone در آن
   * بازه تغییر offset داشته باشد، مقدار
   * نهایی دقیق‌تر شود.
   */
  const secondOffset =
    getTimeZoneOffsetMs(result);

  if (
    secondOffset !== firstOffset
  ) {
    result =
      new Date(
        Date.UTC(
          parts.year,
          parts.month - 1,
          parts.day,
          0,
          0,
          0,
          0,
        ) - secondOffset,
      );
  }

  return result;
}

export function getTodayRange() {
  const now = new Date();

  const todayParts =
    getDateParts(now);

  /*
   * در حالت عادی Intl باید یک تاریخ معتبر
   * برگرداند؛ این fallback فقط برای جلوگیری
   * از انتشار Date نامعتبر در لایه‌های بالاتر
   * گذاشته شده است.
   */
  if (
    !isValidDateParts(
      todayParts,
    )
  ) {
    throw new Error(
      "Unable to resolve today's date for the configured timezone.",
    );
  }

  const tomorrowParts =
    addCalendarDays(
      todayParts,
      1,
    );

  const start =
    getLocalMidnightUtc(
      todayParts,
    );

  const end =
    getLocalMidnightUtc(
      tomorrowParts,
    );

  return {
    start,
    end,
  };
}

export function getDateRange(
  dateKey: string,
) {
  const match =
    /^(\d{4})-(\d{2})-(\d{2})$/.exec(
      dateKey,
    );

  if (!match) {
    return null;
  }

  const year =
    Number(match[1]);

  const month =
    Number(match[2]);

  const day =
    Number(match[3]);

  const parts: DateParts = {
    year,
    month,
    day,
  };

  /*
   * قبلاً فقط بازه 1 تا 31 بررسی می‌شد؛
   * در نتیجه تاریخ‌هایی مثل 2026-02-31
   * می‌توانستند وارد سیستم شوند.
   */
  if (
    !isValidDateParts(parts)
  ) {
    return null;
  }

  const start =
    getLocalMidnightUtc(
      parts,
    );

  const nextDayParts =
    addCalendarDays(
      parts,
      1,
    );

  const end =
    getLocalMidnightUtc(
      nextDayParts,
    );

  return {
    start,
    end,
  };
}

}

function clampMinutes(
  value: number,
): number {
  if (!Number.isFinite(value)) {
    return 0;
  }

  return Math.max(
    0,
    Math.min(
      1439,
      Math.floor(value),
    ),
  );
}

function normalizeSchedule(
  schedule: AttendanceSchedule,
): AttendanceSchedule {
  const start =
    clampMinutes(
      schedule.workStartMinutes,
    );

  const end =
    clampMinutes(
      schedule.workEndMinutes,
    );

  return {
    workStartMinutes: start,
    workEndMinutes: end,

    saturday:
      Boolean(schedule.saturday),

    sunday:
      Boolean(schedule.sunday),

    monday:
      Boolean(schedule.monday),

    tuesday:
      Boolean(schedule.tuesday),

    wednesday:
      Boolean(schedule.wednesday),

    thursday:
      Boolean(schedule.thursday),

    friday:
      Boolean(schedule.friday),
  };
}

function getMinutesOfDay(
  date: Date,
): number {
  const formatter =
    new Intl.DateTimeFormat(
      "en-US",
      {
        timeZone: TIME_ZONE,
        hour: "2-digit",
        minute: "2-digit",
        hourCycle: "h23",
      },
    );

  const parts =
    formatter.formatToParts(date);

  const hour = Number(
    parts.find(
      (part) =>
        part.type === "hour",
    )?.value,
  );

  const minute = Number(
    parts.find(
      (part) =>
        part.type === "minute",
    )?.value,
  );

  if (
    !Number.isFinite(hour) ||
    !Number.isFinite(minute)
  ) {
    return 0;
  }

  return hour * 60 + minute;
}

function getDifferenceInMinutes(
  from: Date,
  to: Date,
): number {
  const difference =
    to.getTime() -
    from.getTime();

  if (difference <= 0) {
    return 0;
  }

  return Math.floor(
    difference / 60000,
  );
}

function isFutureDate(
  workDate: Date,
  now: Date,
): boolean {
  return (
    getDateKey(workDate) >
    getDateKey(now)
  );
}

function normalizeTemporaryExits(
  exits:
    | AttendanceTemporaryExit[]
    | undefined,
): AttendanceTemporaryExit[] {
  if (!exits?.length) {
    return [];
  }

  return exits
    .filter(
      (exit) =>
        exit.startedAt instanceof
        Date,
    )
    .sort(
      (a, b) =>
        a.startedAt.getTime() -
        b.startedAt.getTime(),
    );
}

function getTemporaryExitMinutes(
  attendance: AttendanceRecord | null,
  now: Date,
): number {
  if (
    !attendance?.checkIn
  ) {
    return 0;
  }

  const exits =
    normalizeTemporaryExits(
      attendance.temporaryExits,
    );

  if (!exits.length) {
    return 0;
  }

  const attendanceEnd =
    attendance.checkOut ??
    now;

  const attendanceStart =
    attendance.checkIn;

  let totalMinutes = 0;

  for (const exit of exits) {
    const exitStart =
      exit.startedAt;

    if (
      exitStart.getTime() <=
      attendanceStart.getTime()
    ) {
      continue;
    }

    if (
      exitStart.getTime() >=
      attendanceEnd.getTime()
    ) {
      continue;
    }

    let exitEnd =
      exit.endedAt ??
      null;

    if (
      exit.status ===
        "COMPLETED" &&
      !exitEnd
    ) {
      continue;
    }

    if (!exitEnd) {
      exitEnd =
        now.getTime() <
        attendanceEnd.getTime()
          ? now
          : attendanceEnd;
    }

    const boundedStart =
      exitStart.getTime() >
      attendanceStart.getTime()
        ? exitStart
        : attendanceStart;

    const boundedEnd =
      exitEnd.getTime() <
      attendanceEnd.getTime()
        ? exitEnd
        : attendanceEnd;

    if (
      boundedEnd.getTime() <=
      boundedStart.getTime()
    ) {
      continue;
    }

    totalMinutes +=
      getDifferenceInMinutes(
        boundedStart,
        boundedEnd,
      );
  }

  return Math.max(
    0,
    totalMinutes,
  );
}

function getWorkedMinutes(
  attendance: AttendanceRecord | null,
  now: Date,
): number {
  if (
    !attendance ||
    !attendance.checkIn
  ) {
    return 0;
  }

  const end =
    attendance.checkOut ??
    now;

  const rawMinutes =
    getDifferenceInMinutes(
      attendance.checkIn,
      end,
    );

  const temporaryExitMinutes =
    getTemporaryExitMinutes(
      attendance,
      now,
    );

  return Math.max(
    0,
    rawMinutes -
      temporaryExitMinutes,
  );
}

export function calculateAttendanceDay(
  input: {
    workDate: Date;
    attendance: AttendanceRecord | null;
    schedule: AttendanceSchedule;
    holiday?: AttendanceHoliday | null;
    leave?: AttendanceLeave | null;
    now?: Date;
  },
): AttendanceCalculation {
  const now =
    input.now ?? new Date();

  const schedule =
    normalizeSchedule(
      input.schedule,
    );

  const holiday =
    input.holiday ?? null;

  const leave =
    input.leave ?? null;

  const isHoliday =
    holiday !== null;

  const isLeave =
    leave !== null;

  const isWorkingDay =
    getIsWorkingDay(
      input.workDate,
      schedule,
    );

  const isFuture =
    isFutureDate(
      input.workDate,
      now,
    );

  const attendance =
    input.attendance;

  const hasAttendance =
    attendance !== null;

  const checkIn =
    attendance?.checkIn ??
    null;

  const checkOut =
    attendance?.checkOut ??
    null;

  const isComplete =
    Boolean(
      checkIn &&
        checkOut,
    );

  const scheduledStartMinutes =
    schedule.workStartMinutes;

  const scheduledEndMinutes =
    schedule.workEndMinutes;

  const scheduledMinutes =
    Math.max(
      0,
      scheduledEndMinutes -
        scheduledStartMinutes,
    );

  const temporaryExitMinutes =
    getTemporaryExitMinutes(
      attendance,
      now,
    );

  const workedMinutes =
    getWorkedMinutes(
      attendance,
      now,
    );

  let state: AttendanceDayState;

  /*
   * ترتیب اولویت وضعیت‌ها:
   *
   * ۱. تعطیل
   * ۲. روز غیرکاری
   * ۳. حضور واقعی ثبت‌شده
   * ۴. مرخصی تأییدشده
   * ۵. تاریخ آینده
   * ۶. غیبت
   */
  if (isHoliday) {
    state = "HOLIDAY";
  } else if (!isWorkingDay) {
    state = "NON_WORKING";
  } else if (
    hasAttendance &&
    checkOut
  ) {
    state = "PRESENT";
  } else if (
    hasAttendance &&
    checkIn
  ) {
    state = "WORKING";
  } else if (isLeave) {
    state = "LEAVE";
  } else if (isFuture) {
    state = "FUTURE";
  } else {
    state = "ABSENT";
  }

  let lateMinutes = 0;
  let earlyLeaveMinutes = 0;
  let overtimeMinutes = 0;
  let underworkMinutes = 0;

  /*
   * تأخیر ورود
   *
   * خروج موقت هیچ اثری روی زمان ورود ندارد.
   */
  if (
    checkIn &&
    isWorkingDay &&
    !isHoliday &&
    !isLeave &&
    !isFuture
  ) {
    const actualCheckInMinutes =
      getMinutesOfDay(checkIn);

    if (
      actualCheckInMinutes >
      scheduledStartMinutes
    ) {
      lateMinutes =
        actualCheckInMinutes -
        scheduledStartMinutes;
    }
  }

  /*
   * تعجیل و اضافه‌کاری بر اساس خروج نهایی
   * محاسبه می‌شوند، نه آخرین بازگشت موقت.
   */
  if (
    checkOut &&
    isWorkingDay &&
    !isHoliday &&
    !isLeave &&
    !isFuture
  ) {
    const actualCheckOutMinutes =
      getMinutesOfDay(checkOut);

    if (
      actualCheckOutMinutes <
      scheduledEndMinutes
    ) {
      earlyLeaveMinutes =
        scheduledEndMinutes -
        actualCheckOutMinutes;
    }

    if (
      actualCheckOutMinutes >
      scheduledEndMinutes
    ) {
      overtimeMinutes =
        actualCheckOutMinutes -
        scheduledEndMinutes;
    }
  }

  /*
   * کم‌کاری بر اساس کارکرد واقعی
   * بعد از کسر خروج‌های موقت محاسبه می‌شود.
   */
  if (
    isComplete &&
    isWorkingDay &&
    !isHoliday &&
    !isLeave &&
    !isFuture &&
    workedMinutes <
      scheduledMinutes
  ) {
    underworkMinutes =
      scheduledMinutes -
      workedMinutes;
  }

  /*
   * تراز نیز بر اساس کارکرد واقعی است.
   */
  const balanceMinutes =
    isComplete &&
    isWorkingDay &&
    !isHoliday &&
    !isLeave &&
    !isFuture
      ? workedMinutes -
        scheduledMinutes
      : 0;

  return {
    state,

    isWorkingDay,
    isHoliday,
    isFuture,
    isLeave,

    holiday,
    leave,

    scheduledStartMinutes,
    scheduledEndMinutes,
    scheduledMinutes,

    checkIn,
    checkOut,

    workedMinutes,
    temporaryExitMinutes,

    lateMinutes,
    earlyLeaveMinutes,
    overtimeMinutes,
    underworkMinutes,

    balanceMinutes,

    hasAttendance,
    isComplete,

    attendanceStatus:
      attendance?.status ??
      null,

    note:
      attendance?.note ??
      null,
  };
}

export function minutesToClock(
  totalMinutes: number,
): string {
  const safe =
    Number.isFinite(
      totalMinutes,
    )
      ? Math.max(
          0,
          Math.floor(
            totalMinutes,
          ),
        )
      : 0;

  const hours =
    Math.floor(
      safe / 60,
    );

  const minutes =
    safe % 60;

  return `${String(hours).padStart(
    2,
    "0",
  )}:${String(minutes).padStart(
    2,
    "0",
  )}`;
}

export function formatAttendanceMinutes(
  totalMinutes: number,
): string {
  const safe =
    Number.isFinite(
      totalMinutes,
    )
      ? Math.max(
          0,
          Math.floor(
            totalMinutes,
          ),
        )
      : 0;

  const hours =
    Math.floor(
      safe / 60,
    );

  const minutes =
    safe % 60;

  if (hours === 0) {
    return `${minutes.toLocaleString(
      "fa-IR",
    )} دقیقه`;
  }

  if (minutes === 0) {
    return `${hours.toLocaleString(
      "fa-IR",
    )} ساعت`;
  }

  return `${hours.toLocaleString(
    "fa-IR",
  )} ساعت و ${minutes.toLocaleString(
    "fa-IR",
  )} دقیقه`;
}

export function formatSignedAttendanceMinutes(
  totalMinutes: number,
): string {
  if (
    !Number.isFinite(
      totalMinutes,
    ) ||
    totalMinutes === 0
  ) {
    return "۰ دقیقه";
  }

  const absolute =
    Math.abs(
      Math.floor(
        totalMinutes,
      ),
    );

  const formatted =
    formatAttendanceMinutes(
      absolute,
    );

  return totalMinutes > 0
    ? `+${formatted}`
    : `-${formatted}`;
}

/**
 * از بین انتساب‌های موجود، آخرین انتسابی که
 * در تاریخ موردنظر معتبر است را انتخاب می‌کند.
 *
 * startsAt: inclusive
 * endsAt: exclusive
 */
export function pickEmployeeShift(
  assignments: EmployeeShiftAssignment[],
  userId: string,
  workDate: Date,
): EmployeeShiftAssignment | null {
  const matching = assignments
    .filter(
      (assignment) =>
        assignment.userId === userId &&
        assignment.startsAt.getTime() <=
          workDate.getTime() &&
        (assignment.endsAt === null ||
          assignment.endsAt.getTime() >
            workDate.getTime()),
    )
    .sort(
      (a, b) =>
        b.startsAt.getTime() -
        a.startsAt.getTime(),
    );

  return matching[0] ?? null;
}

/**
 * برنامه واقعی کارمند را برای یک تاریخ مشخص
 * تعیین می‌کند:
 *
 * 1. شیفت اختصاصی معتبر
 * 2. در غیر این صورت برنامه پیش‌فرض شرکت
 */
export function resolveEmployeeSchedule(
  params: {
    assignments: EmployeeShiftAssignment[];
    userId: string;
    workDate: Date;
    fallbackSchedule: AttendanceSchedule;
  },
): AttendanceSchedule {
  const assignment = pickEmployeeShift(
    params.assignments,
    params.userId,
    params.workDate,
  );

  if (!assignment) {
    return params.fallbackSchedule;
  }

  return shiftToAttendanceSchedule(
    assignment.shift,
  );
}

function isPrismaKnownRequestError(
  error: unknown,
): error is {
  code: string;
} {
  return (

        code?: unknown;
      }
    ).code === "string"
  );

function normalizeOptionalText(
  value: FormDataEntryValue | null | undefined,
  maxLength: number,
): string | null {
  if (typeof value !== "string") {
    return null;
  }

  const normalized =
    value.trim();

  if (!normalized) {
    return null;
  }

  return normalized.slice(
    0,
    maxLength,
  );
}

export async function checkIn(
  _previousState: AttendanceActionState = emptyState,
  _formData?: FormData,
): Promise<AttendanceActionState> {
  const user =
    await requireEmployee();

  const { start, end } =
    getTodayRange();

  try {
    const [
      existing,
      settings,
      todayHoliday,
    ] = await Promise.all([
      prisma.attendance.findUnique({
        where: {
          userId_workDate: {
            userId: user.id,
            workDate: start,
          },
        },
        select: {
          id: true,
          checkIn: true,
          checkOut: true,
        },
      }),

      prisma.companySettings.findUnique({
        where: {
          id: "company",
        },
      }),

      prisma.holiday.findUnique({
        where: {
          date: start,
        },
        select: {
          title: true,
          description: true,
        },
      }),
    ]);

    const schedule =
      settings ?? DEFAULT_SCHEDULE;

    const dayCalculation =
      calculateAttendanceDay({
        workDate: start,
        schedule,
        holiday:
          todayHoliday
            ? {
                title:
                  todayHoliday.title,
                description:
                  todayHoliday.description,
              }
            : null,
        attendance: null,
      });

    if (
      dayCalculation.state ===
      "HOLIDAY"
    ) {
      return {
        success: false,
        message:
          todayHoliday?.title
            ? `امروز تعطیل است: ${todayHoliday.title}`
            : "امروز تعطیل است و امکان ثبت ورود وجود ندارد.",
      };
    }

    if (
      dayCalculation.state ===
      "NON_WORKING"
    ) {
      return {
        success: false,
        message:
          "امروز طبق برنامه کاری شرکت روز غیرکاری است.",
      };
    }

    if (existing?.checkIn) {
      return {
        success: false,
        message:
          "ورود امروز شما قبلاً ثبت شده است.",
      };
    }

    if (
      existing &&
      !existing.checkIn &&
      existing.checkOut
    ) {
      return {
        success: false,
        message:
          "رکورد حضور امروز وضعیت نامعتبر دارد. لطفاً با مدیر سیستم تماس بگیرید.",
      };
    }

    if (existing) {
      const result =
        await prisma.attendance.updateMany({
          where: {
            id: existing.id,
            checkIn: null,
            checkOut: null,
          },
          data: {
            checkIn: new Date(),
            status: "OPEN",
          },
        });

      if (result.count === 0) {
        return {
          success: false,
          message:
            "ورود امروز شما قبلاً ثبت شده است.",
        };
      }
    } else {
      try {
        await prisma.attendance.create({
          data: {
            userId: user.id,
            workDate: start,
            checkIn: new Date(),
            status: "OPEN",
          },
        });
      } catch (error) {
        if (
          isPrismaKnownRequestError(
            error,
          ) &&
          error.code === "P2002"
        ) {
          return {
            success: false,
            message:
              "ورود امروز شما قبلاً ثبت شده است.",
          };
        }

        throw error;
      }
    }

    revalidatePath("/employee");
    revalidatePath("/admin");
    revalidatePath("/admin/attendance");
    revalidatePath(
      "/admin/attendance/[userId]",
    );
    revalidatePath("/admin/calendar");
    revalidatePath("/admin/reports");

    return {
      success: true,
      message:
        "ورود شما با موفقیت ثبت شد.",
    };
  } catch (error) {
    console.error(
      "Employee check-in error:",
      error,
    );

    return {
      success: false,
      message:
        "ثبت ورود انجام نشد. دوباره تلاش کنید.",
    };
  }
}

export async function startTemporaryExit(
  _previousState: TemporaryExitActionState = temporaryExitEmptyState,
  formData?: FormData,
): Promise<TemporaryExitActionState> {
  const user =
    await requireEmployee();

  const { start } =
    getTodayRange();

  try {
    const type =
      parseTemporaryExitType(
        String(
          formData?.get("type") ??
            "",
        ),
      );

    if (!type) {
      return {
        success: false,
        message:
          "نوع خروج را انتخاب کنید.",
      };
    }

    const reason =
      normalizeOptionalText(
        formData?.get("reason"),
        500,
      );

    const result =
      await prisma.$transaction(
        async (tx) => {
          const attendance =
            await tx.attendance.findUnique({
              where: {
                userId_workDate: {
                  userId: user.id,
                  workDate: start,
                },
              },
              select: {
                id: true,
                checkIn: true,
                checkOut: true,
              },
            });

          if (!attendance?.checkIn) {
            return {
              ok: false as const,
              message:
                "ابتدا باید ورود خود را ثبت کنید.",
            };
          }

          if (attendance.checkOut) {
            return {
              ok: false as const,
              message:
                "حضور امروز شما قبلاً با خروج نهایی بسته شده است.",
            };
          }

          /*
           * با تغییر رکورد Attendance در ابتدای تراکنش،
           * ردیف همان روز قفل می‌شود و دو درخواست
           * همزمان نمی‌توانند برای یک کارمند، دو
           * خروج باز ایجاد کنند.
           */
          await tx.attendance.update({
            where: {
              id: attendance.id,
            },
            data: {
              updatedAt: new Date(),
            },
          });

          const openExit =
            await tx.temporaryExit.findFirst({
              where: {
                attendanceId:
                  attendance.id,
                status: "OPEN",
              },
              select: {
                id: true,
              },
            });

          if (openExit) {
            return {
              ok: false as const,
              message:
                "در حال حاضر یک خروج باز دارید. ابتدا باید بازگشت خود را ثبت کنید.",
            };
          }

          const temporaryExit =
            await tx.temporaryExit.create({
              data: {
                userId: user.id,
                attendanceId:
                  attendance.id,
                type,
                status: "OPEN",
                startedAt:
                  new Date(),
                reason,
              },
              select: {
                id: true,
                type: true,
                status: true,
                startedAt: true,
                endedAt: true,
                reason: true,
                note: true,
              },
            });

          await tx.temporaryExitAudit.create({
            data: {
              temporaryExitId:
                temporaryExit.id,

              actorId:
                user.id,

              action: "CREATE",

              previousType:
                null,

              previousStatus:
                null,

              previousStartedAt:
                null,

              previousEndedAt:
                null,

              previousReason:
                null,

              previousNote:
                null,

              newType:
                temporaryExit.type,

              newStatus:
                temporaryExit.status,

              newStartedAt:
                temporaryExit.startedAt,

              newEndedAt:
                temporaryExit.endedAt,

              newReason:
                temporaryExit.reason,

              newNote:
                temporaryExit.note,
            },
          });

          return {
            ok: true as const,
            message:
              type === "MISSION"
                ? "مأموریت شما با موفقیت ثبت شد."
                : "خروج موقت شما با موفقیت ثبت شد.",
          };
        },
      );

    if (!result.ok) {
      return {
        success: false,
        message: result.message,
      };
    }

    revalidatePath("/employee");
    revalidatePath("/admin");
    revalidatePath("/admin/attendance");
    revalidatePath(
      "/admin/attendance/[userId]",
    );
    revalidatePath("/admin/calendar");
    revalidatePath("/admin/reports");

    return {
      success: true,
      message: result.message,
    };
  } catch (error) {
    console.error(
      "Employee temporary exit start error:",
      error,
    );

    return {
      success: false,
      message:
        "ثبت خروج موقت انجام نشد. دوباره تلاش کنید.",
    };
  }
}

  const user =
    await requireEmployee();

  const { start } =
    getTodayRange();

  try {
    const result =
      await prisma.$transaction(
        async (tx) => {

                  workDate: start,

                attendanceId:

                id: true,

            };
          }

          const result =
            await tx.attendance.updateMany({

                id: attendance.id,

                },
                checkOut: null,
              },

                status:
                  "COMPLETED",

            };

    revalidatePath("/admin");
    revalidatePath("/admin/attendance");

"use client";

import {
  useEffect,
  useState,
} from "react";

function formatDuration(
  milliseconds: number,
): string {
  const totalMinutes = Math.max(
    0,
    Math.floor(
      milliseconds / 60000,
    ),
  );

  const hours = Math.floor(
    totalMinutes / 60,
  );

  const minutes =
    totalMinutes % 60;

  const seconds = Math.floor(
    (milliseconds % 60000) / 1000,
  );

  if (hours > 0) {
    return `${hours.toLocaleString(
      "fa-IR",
    )} ساعت و ${minutes.toLocaleString(
      "fa-IR",
    )} دقیقه`;
  }

  if (minutes > 0) {
    return `${minutes.toLocaleString(
      "fa-IR",
    )} دقیقه و ${seconds.toLocaleString(
      "fa-IR",
    )} ثانیه`;
  }

  return `${seconds.toLocaleString(
    "fa-IR",
  )} ثانیه`;
}

export default function TemporaryExitDuration({
  startedAt,
}: {
  startedAt: string;
}) {
  const [now, setNow] =
    useState<Date | null>(null);

  useEffect(() => {
    const current = new Date();

    setNow(current);

    const interval =
      window.setInterval(() => {
        setNow(new Date());
      }, 1000);

    return () =>
      window.clearInterval(
        interval,
      );
  }, []);

  const start =
    new Date(startedAt);

  const duration =
    now && Number.isFinite(
      start.getTime(),
    )
      ? formatDuration(
          now.getTime() -
            start.getTime(),
        )
      : "در حال محاسبه...";

  return (
    <span className="font-semibold text-amber-700">
      {duration}
    </span>
  );
}

import { NextResponse } from "next/server";
import { z } from "zod";

import { verifyPassword } from "@/lib/auth/password";
import { createSession } from "@/lib/auth/session";
import { prisma } from "@/lib/prisma";

const loginSchema = z.object({
  username: z
    .string()
    .trim()
    .min(1, "نام کاربری الزامی است.")
    .max(100),

  password: z
    .string()
    .min(1, "رمز عبور الزامی است.")
    .max(200),
});

export async function POST(request: Request) {
  try {
    const body = await request.json();

    const validation = loginSchema.safeParse(body);

    if (!validation.success) {
      return NextResponse.json(
        {
          success: false,
          message: "اطلاعات ورود صحیح نیست.",
        },
        {
          status: 400,
        },
      );
    }

    const {
      username,
      password,
    } = validation.data;

    const user = await prisma.user.findUnique({
      where: {
        username,
      },
    });

    if (!user || !user.isActive) {
      return NextResponse.json(
        {
          success: false,
          message:
            "نام کاربری یا رمز عبور اشتباه است.",
        },
        {
          status: 401,
        },
      );
    }

    const validPassword =
      await verifyPassword(
        password,
        user.passwordHash,
      );

    if (!validPassword) {
      return NextResponse.json(
        {
          success: false,
          message:
            "نام کاربری یا رمز عبور اشتباه است.",
        },
        {
          status: 401,
        },
      );
    }

    await createSession(user.id);

    const redirectTo =
      user.role === "ADMIN"
        ? "/admin"
        : "/employee";

    return NextResponse.json({
      success: true,

      redirectTo,

      user: {
        id: user.id,
        username: user.username,
        firstName: user.firstName,
        lastName: user.lastName,
        role: user.role,
      },
    });
  } catch (error) {
    console.error("Login error:", error);

    return NextResponse.json(
      {
        success: false,
        message: "خطایی در ورود رخ داد.",
      },
      {
        status: 500,
      },
    );
  }
}

function formatMinutes(
  totalMinutes: number,
): string {
  if (totalMinutes <= 0) {
    return "۰ دقیقه";
  }

  const hours = Math.floor(
    totalMinutes / 60,
  );

  const minutes =
    totalMinutes % 60;

  if (hours === 0) {
    return `${minutes} دقیقه`;
  }

  if (minutes === 0) {
    return `${hours} ساعت`;
  }

  return `${hours} ساعت و ${minutes} دقیقه`;
}

    checkIn: null,
    checkOut: null,
    workedMinutes: 0,
    scheduledMinutes: 0,
    lateMinutes: 0,
    earlyLeaveMinutes: 0,
    overtimeMinutes: 0,
    underworkMinutes: 0,
    balanceMinutes: 0,
    hasAttendance: false,
    isComplete: false,
    attendanceStatus: null,
    note: null,
  };
}

function leaveCoversDate(
  leave: LeaveRecord,
  dateKey: string,
): boolean {
  const range =
    getDateRange(dateKey);

  if (!range) {
    return false;
  }

  return (
    leave.startDate < range.end &&
    leave.endDate >= range.start
  );
}

function getLeaveForDate(
  leaves: LeaveRecord[],
  dateKey: string,
): LeaveRecord | null {
  return (
    leaves.find((leave) =>
      leaveCoversDate(
        leave,
        dateKey,
      ),
    ) ?? null
  );
}

function isApprovedLeaveDay(
  calculation: AttendanceCalculation,
  attendance: {
    checkIn: Date | null;
    checkOut: Date | null;
  } | null,
  leave: LeaveRecord | null,
): boolean {
  return Boolean(
    leave &&
      !attendance &&
      calculation.isWorkingDay &&
      !calculation.isHoliday &&
      !calculation.isFuture,
  );
}
